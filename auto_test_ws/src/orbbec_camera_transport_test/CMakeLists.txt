cmake_minimum_required(VERSION 3.16)
project(orbbec_camera_transport_test)

find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_components REQUIRED)
find_package(sensor_msgs REQUIRED)

set(TRANSPORT_GPU_BACKEND "NONE" CACHE STRING "GPU backend for the transport test subscriber")
set(transport_dependencies rclcpp rclcpp_components sensor_msgs)
if(TRANSPORT_GPU_BACKEND STREQUAL "CUDA_BUFFER")
  list(APPEND transport_dependencies cuda_buffer rosidl_buffer rosidl_buffer_backend_registry)
elseif(TRANSPORT_GPU_BACKEND MATCHES "^NITROS_(GXF|BUFFER)$")
  list(APPEND transport_dependencies
    isaac_ros_nitros isaac_ros_nitros_image_type isaac_ros_nitros_point_cloud_type)
  if(TRANSPORT_GPU_BACKEND STREQUAL "NITROS_GXF")
    list(APPEND transport_dependencies
      isaac_ros_gxf negotiated gxf_isaac_gems gxf_isaac_ros_messages)
  endif()
elseif(NOT TRANSPORT_GPU_BACKEND STREQUAL "NONE")
  message(FATAL_ERROR "Unsupported transport GPU backend: ${TRANSPORT_GPU_BACKEND}")
endif()

add_library(transport_subscriber_component SHARED src/transport_subscriber.cpp)
target_compile_features(transport_subscriber_component PRIVATE cxx_std_17)
target_link_libraries(transport_subscriber_component ${CMAKE_DL_LIBS})
if(NOT TRANSPORT_GPU_BACKEND STREQUAL "NONE")
  find_package(CUDAToolkit REQUIRED)
  foreach(dependency IN LISTS transport_dependencies)
    find_package(${dependency} REQUIRED)
  endforeach()
  target_compile_definitions(transport_subscriber_component PRIVATE TRANSPORT_${TRANSPORT_GPU_BACKEND})
  target_link_libraries(transport_subscriber_component CUDA::cudart CUDA::cuda_driver)
  set_target_properties(transport_subscriber_component PROPERTIES INSTALL_RPATH_USE_LINK_PATH TRUE)
endif()
if(COMMAND ament_target_dependencies)
  ament_target_dependencies(transport_subscriber_component ${transport_dependencies})
else()
  target_link_libraries(transport_subscriber_component rclcpp::rclcpp rclcpp_components::component
    sensor_msgs::sensor_msgs_library)
  foreach(dependency IN LISTS transport_dependencies)
    if(NOT dependency MATCHES "^(rclcpp|rclcpp_components|sensor_msgs)$")
      target_link_libraries(transport_subscriber_component ${dependency}::${dependency})
    endif()
  endforeach()
endif()
rclcpp_components_register_node(transport_subscriber_component
  PLUGIN "orbbec_camera_transport_test::TransportSubscriber"
  EXECUTABLE transport_subscriber)
install(TARGETS transport_subscriber_component LIBRARY DESTINATION lib)
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME} FILES_MATCHING PATTERN "*.launch.py")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/ros_distro.txt" "$ENV{ROS_DISTRO}\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/gpu_backend.txt" "${TRANSPORT_GPU_BACKEND}\n")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/ros_distro.txt"
  "${CMAKE_CURRENT_BINARY_DIR}/gpu_backend.txt" DESTINATION share/${PROJECT_NAME})
ament_package()
